func main() -> int {
    var score : int = 75;

    if (score >= 90) {
        print(1);
    } else if (score >= 75) {
        print(2);
    } else if (score >= 60) {
        print(3);
    } else {
        print(4);
    }

    var flag : bool = true;
    if (flag) {
        print(100);
    }

    var n : int = 7;
    if (n == 0) {
        print(0);
    } else {
        print(n);
    }
}
