struct Arith {
    func double(x : int) -> int {
        return x * 2;
    }

    func add(a : int, b : int) -> int {
        return a + b;
    }
}

func main() -> int {
    var a : Arith = new Arith();

    var r : int = a.double(5) + a.add(1, 2);
    print(r);

    if (a.double(3) == 6) {
        print(1);
    }

    a.double(99);
}
